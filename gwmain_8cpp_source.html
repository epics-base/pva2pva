68void getargs(ServerConfig& arg, int argc, char *argv[])
69{
70 int opt;
71 bool checkonly = false;
72
73 while( (opt=getopt(argc, argv, "qvhiIC"))!=-1)
74 {
75 switch(opt) {
76 case 'h':
77 usage(argv[0]);
78 exit(0);
79 case 'q':
80 arg.debug--;
81 break;
82 case 'v':
83 arg.debug++;
84 break;
85 case 'I':
86 arg.interactive = true;
87 break;
88 case 'i':
89 arg.interactive = false;
90 break;
91 case 'C':
92 checkonly = true;
93 break;
94 default:
95 std::cerr<<"Unknown argument -"<<char(opt)<<"\n";
96 usage(argv[0]);
97 exit(1);
98 }
99 }
100
101 if(optind!=argc-1) {
102 std::cerr<<"Exactly one positional argument expected\n";
103 usage(argv[0]);
104 exit(1);
105 }
106
107 arg.conf = pvd::getPVDataCreate()->createPVStructure(schema);
108 std::ifstream strm(argv[optind]);
109 pvd::parseJSON(strm, arg.conf);
110
111 p2pReadOnly = arg.conf->getSubFieldT<pvd::PVScalar>("readOnly")->getAs<pvd::boolean>();
112
113 unsigned version = arg.conf->getSubFieldT<pvd::PVUInt>("version")->get();
114 if(version==0) {
115 std::cerr<<"Warning: config file missing \"version\" key. Assuming 1\n";
116 } else if(version!=1) {
117 std::cerr<<"config file version mis-match. expect 1 found "<<version<<"\n";
118 exit(1);
119 }
120 if(arg.conf->getSubFieldT<pvd::PVStructureArray>("clients")->view().empty()) {
121 std::cerr<<"No clients configured\n";
122 exit(1);
123 }
124 if(arg.conf->getSubFieldT<pvd::PVStructureArray>("servers")->view().empty()) {
125 std::cerr<<"No servers configured\n";
126 exit(1);
127 }
128
129 if(checkonly) {
130 std::cerr<<"Config file OK\n";
131 exit(0);
132 }
133}

135GWServerChannelProvider::shared_pointer configure_client(ServerConfig& arg, const pvd::PVStructurePtr& conf)
136{
137 std::string name(conf->getSubFieldT<pvd::PVString>("name")->get());
138 std::string provider(conf->getSubFieldT<pvd::PVString>("provider")->get());
139
140 LOG(pva::logLevelInfo, "Configure client '%s' with provider '%s'", name.c_str(), provider.c_str());
141
142 pva::Configuration::shared_pointer C(pva::ConfigurationBuilder()
143 .add("EPICS_PVA_ADDR_LIST", conf->getSubFieldT<pvd::PVString>("addrlist")->get())
144 .add("EPICS_PVA_AUTO_ADDR_LIST", conf->getSubFieldT<pvd::PVScalar>("autoaddrlist")->getAs<std::string>())
145 .add("EPICS_PVA_SERVER_PORT", conf->getSubFieldT<pvd::PVScalar>("serverport")->getAs<pvd::uint16>())
146 .add("EPICS_PVA_BROADCAST_PORT", conf->getSubFieldT<pvd::PVScalar>("bcastport")->getAs<pvd::uint16>())
147 .add("EPICS_PVA_DEBUG", arg.debug>=5 ? 5 : 0)
148 .push_map()
149 .build());
150
151 pva::ChannelProvider::shared_pointer base(pva::ChannelProviderRegistry::clients()->createProvider(provider, C));
152 if(!base)
153 throw std::runtime_error("Can't create ChannelProvider");
154
155 GWServerChannelProvider::shared_pointer ret(new GWServerChannelProvider(base));
156 return ret;
157}

159pva::ServerContext::shared_pointer configure_server(ServerConfig& arg, const pvd::PVStructurePtr& conf)
160{
161 std::string name(conf->getSubFieldT<pvd::PVString>("name")->get());
162
163 LOG(pva::logLevelInfo, "Configure server '%s'", name.c_str());
164
165 pva::Configuration::shared_pointer C(pva::ConfigurationBuilder()
166 .add("EPICS_PVAS_INTF_ADDR_LIST", conf->getSubFieldT<pvd::PVString>("interface")->get())
167 .add("EPICS_PVAS_BEACON_ADDR_LIST", conf->getSubFieldT<pvd::PVString>("addrlist")->get())
168 .add("EPICS_PVAS_AUTO_BEACON_ADDR_LIST", conf->getSubFieldT<pvd::PVScalar>("autoaddrlist")->getAs<std::string>())
169 .add("EPICS_PVAS_SERVER_PORT", conf->getSubFieldT<pvd::PVScalar>("serverport")->getAs<pvd::uint16>())
170 .add("EPICS_PVAS_BROADCAST_PORT", conf->getSubFieldT<pvd::PVScalar>("bcastport")->getAs<pvd::uint16>())
171 .add("EPICS_PVA_DEBUG", arg.debug>=5 ? 5 : 0)
172 .push_map()
173 .build());
174
175 pvd::PVStringArray::shared_pointer clients(conf->getSubFieldT<pvd::PVStringArray>("clients"));
176 pvd::PVStringArray::const_svector names(clients->view());
177 std::vector<pva::ChannelProvider::shared_pointer> providers;
178
179 for(pvd::PVStringArray::const_svector::const_iterator it(names.begin()), end(names.end()); it!=end; ++it)
180 {
181 ServerConfig::clients_t::const_iterator it2(arg.clients.find(*it));
182 if(it2==arg.clients.end())
183 throw std::runtime_error("Server references non-existant client");
184 providers.push_back(it2->second);
185 }
186
187 pva::ServerContext::shared_pointer ret(pva::ServerContext::create(pva::ServerContext::Config()
188 .config(C)
189 .providers(providers)));
190 return ret;
191}
